<script>
// ===== Pengganti google.script.run -> memanggil Apps Script lewat fetch (dengan token sesi) =====
(function () {
  var API_URL = '';   // WAJIB DIISI: URL Web App Apps Script yang berakhiran /exec
  var KEY = 'WMS_SESSION_V1';

  window.WMS_getSession = function () {
    try { return JSON.parse(localStorage.getItem(KEY) || 'null'); } catch (e) { return null; }
  };
  window.WMS_saveSession = function (o) {
    try { localStorage.setItem(KEY, JSON.stringify(o)); } catch (e) {}
  };
  window.WMS_clearSession = function () {
    try { localStorage.removeItem(KEY); } catch (e) {}
  };

  window.wmsToast = function (msg, isErr) {
    var t = document.getElementById('toast'), m = document.getElementById('toastMsg');
    if (!t || !m) { alert(msg); return; }
    m.textContent = (isErr ? '\u26A0 ' : '') + msg;
    t.classList.add('show');
    clearTimeout(window._wmsToastTimer);
    window._wmsToastTimer = setTimeout(function () { t.classList.remove('show'); }, 4000);
  };
  // penanganan bawaan jika sebuah panggilan gagal dan tampilan tidak menyediakan handler sendiri
  window.WMS_onApiError = function (msg) { window.wmsToast(msg, true); };

  function handleError(j, fail) {
    var err = { message: (j && j.error) || 'Respon server tidak valid', code: (j && j.code) || '' };
    if (err.code === 'AUTH') {
      window.WMS_clearSession();
      if (window.WMS_onSessionExpired) window.WMS_onSessionExpired(err.message);
      return;
    }
    if (err.code === 'MUST_CHANGE') {
      if (window.WMS_onMustChange) window.WMS_onMustChange();
      return;
    }
    if (fail) fail(err);
    else if (window.WMS_onApiError) window.WMS_onApiError(err.message);
  }

  function call(fn, args, ok, fail) {
    if (!API_URL) {
      setTimeout(function () { handleError({ error: 'API_URL belum diisi di index.html' }, fail); }, 0);
      return;
    }
    var sess = window.WMS_getSession();
    var ctrl = (typeof AbortController !== 'undefined') ? new AbortController() : null;
    var timer = setTimeout(function () { if (ctrl) ctrl.abort(); }, 90000);
    fetch(API_URL, {
      method: 'POST',
      body: JSON.stringify({ fn: fn, args: args, token: (sess && sess.token) || '' }),
      signal: ctrl ? ctrl.signal : undefined
    })
      .then(function (r) { return r.json(); })
      .then(function (j) {
        clearTimeout(timer);
        if (j && j.ok) { if (ok) ok(j.data); }
        else handleError(j, fail);
      })
      .catch(function (e) {
        clearTimeout(timer);
        handleError({ error: (e && e.name === 'AbortError') ? 'Server terlalu lama merespons. Coba lagi.' : 'Gagal menghubungi server. Periksa koneksi internet.' }, fail);
      });
  }

  function makeRunner(ok, fail) {
    return new Proxy({}, {
      get: function (_, prop) {
        if (prop === 'withSuccessHandler') return function (f) { return makeRunner(f, fail); };
        if (prop === 'withFailureHandler') return function (f) { return makeRunner(ok, f); };
        return function () { call(String(prop), Array.prototype.slice.call(arguments), ok, fail); };
      }
    });
  }

  window.google = window.google || {};
  window.google.script = { run: makeRunner(null, null) };
})();
</script>

function restoreSession(){
  const sess = WMS_getSession();
  if(!sess || !sess.token) return;
  const msgEl = document.getElementById('loginMsg');
  msgEl.textContent = 'Memulihkan sesi...';
  google.script.run.withSuccessHandler(res=>{

function renderBakuTable(container, data, statusType, tableId, fbCounts, ffMap){
  if(!data || data.length===0){ container.innerHTML = `<div class="empty-state"><p>Tidak ada data ditemukan.</p></div>`; icons(); return; }
  const pillClass = statusType==='SO' ? 'pill-so' : (statusType==='SO OUT' ? 'pill-out' : 'pill-retur');
  const isSoOut = (statusType === 'SO OUT');
  const isRetur = (statusType === 'RETUR');
  fbCounts = fbCounts || {}; ffMap = ffMap || {};

  const isSelected = (val, target) => {
    if (!target) return '';
    return String(val).trim().toLowerCase() === String(target).trim().toLowerCase() ? 'selected' : '';
  };

  let rows = data.map(d=>{
    let stockNum = Number(d.stock); let stockBadgeStyle = ''; let stockText = d.stock;
    if (isNaN(stockNum) || stockNum <= 0) { stockBadgeStyle = 'background:#fee2e2;color:var(--danger);'; stockText = 'STOCK KOSONG'; } 
    else if (stockNum <= 5) { stockBadgeStyle = 'background:#fef3c7;color:var(--warn);'; } 
    else { stockBadgeStyle = 'background:#d1fae5;color:var(--ok);'; }

    const invUpper = String(d.noInvoice || '').trim().toUpperCase();
    const chatCount = fbCounts[invUpper] || 0;
    const badgeDisplay = chatCount > 0 ? `display:flex;` : `display:none;`;
    const currentFF = String(ffMap[invUpper] || '').trim();
    
    const courierStr = String(d.courier || '-').toUpperCase();
    let courierBadgeStyle = courierStr.includes('INSTAN') ? 'background:#d1fae5;color:#059669;' : 'background:#fef3c7;color:#d97706;';

    let rowBgStyle = ''; let tglDisplay = escapeHtml(d.tgl);
    if (statusType === 'SO' && d.agingDays !== undefined) {
      if (d.agingDays === 1) {
        rowBgStyle = 'background-color: #fef9c3;'; 
        tglDisplay += `<br><span style="font-size:10px;font-weight:800;color:#ca8a04;background:#fef08a;padding:2px 6px;border-radius:4px;margin-top:5px;display:inline-block;">Endap 1 Hari</span>`;
      } else if (d.agingDays >= 2) {
        rowBgStyle = 'background-color: #fee2e2;';
        tglDisplay += `<br><span style="font-size:10px;font-weight:800;color:#991b1b;background:#fca5a5;padding:2px 6px;border-radius:4px;margin-top:5px;display:inline-block;">Tertahan ${d.agingDays} Hari!</span>`;
      }
    }

    return `
    <tr class="baku-row" style="${rowBgStyle}">
      <td><span class="pill ${pillClass}">${escapeHtml(d.status)}</span></td>
      <td>${tglDisplay}</td>
      <td>${escapeHtml(d.time)}</td>
      <td><b>${escapeHtml(d.team)}</b></td>
      <td><span class="pill" style="${courierBadgeStyle}; font-size:11px;">${escapeHtml(courierStr)}</span></td>
      ${isSoOut ? `<td><b>${escapeHtml(d.kodeBarang)}</b></td><td><b style="color:var(--primary);">${escapeHtml(d.noInvoice)}</b></td>` : `<td style="color:var(--primary);font-weight:700;">${escapeHtml(d.noInvoice)}</td><td>${escapeHtml(d.kodeBarang)}</td>`}
      <td>${escapeHtml(String(d.qty))}</td>
      ${isSoOut || isRetur ? '' : `<td><span class="badge" style="${stockBadgeStyle}padding:4px 10px;border-radius:6px;font-weight:700;display:inline-block;font-size:11.5px;">${escapeHtml(stockText)}</span></td>`}
      ${isRetur ? `<td><b>${escapeHtml(d.notaRetur)}</b></td>` : ''}
      ${isSoOut || isRetur ? '' : `<td><button class="icon-btn" onclick='openChat(${JSON.stringify(d.noInvoice)})'><i data-lucide="message-circle" style="width:16px;height:16px;"></i><span class="mini-badge fb-badge" style="${badgeDisplay}">${chatCount}</span></button></td>`}
      ${isSoOut || isRetur ? '' : `<td>
        <select onchange="updateFastFeedback('${escapeHtml(d.noInvoice)}', this.value)" style="padding:6px 10px;border-radius:8px;border:1px solid var(--border);font-size:11.5px;background:#fff;outline:none;">
          <option value="">-- Pilih Fast Feedback --</option>
          <option value="Tunggu Unit Lain" ${isSelected('Tunggu Unit Lain', currentFF)}>Tunggu Unit Lain</option>
          <option value="Stock Adjustment" ${isSelected('Stock Adjustment', currentFF)}>Stock Adjustment</option>
          <option value="Lewat Jam Cutoff" ${isSelected('Lewat Jam Cutoff', currentFF)}>Lewat Jam Cutoff</option>
          <option value="Stock Kosong" ${isSelected('Stock Kosong', currentFF)}>Stock Kosong</option>
        </select>
      </td>`}
      <td><button class="icon-btn" onclick='openTracking(${JSON.stringify(d.noInvoice)}, ${JSON.stringify(statusType)})'><i data-lucide="route" style="width:16px;height:16px;"></i></button></td>
    </tr>`;
  }).join('');

  let headersHtml = `<tr><th>STATUS</th><th>TGL</th><th>TIME</th><th>TEAM</th><th>KURIR</th><th>${isSoOut ? 'SKU' : 'NO INVOICE'}</th><th>${isSoOut ? 'NO INVOICE' : 'KODE BARANG'}</th><th>QTY</th>${isSoOut || isRetur ? '' : '<th>STOCK</th>'}${isRetur ? '<th>NOTA RETUR</th>' : ''}${isSoOut || isRetur ? '' : '<th>FEEDBACK</th><th>FAST FEEDBACK</th>'}<th>HISTORY</th></tr>`;

  let filterInputsHtml = `
  <tr class="filter-row">
    <th><input type="text" placeholder="Cari..." oninput="applyColumnFilter('${tableId}', 0, this.value)"></th>
    <th><input type="text" placeholder="Cari..." oninput="applyColumnFilter('${tableId}', 1, this.value)"></th>

    ${isSoOut || isRetur ? '' : '<th></th><th></th>'}
    <th></th>
  </tr>`;

  container.innerHTML = `<table class="table-baku" id="tbl-${tableId}"><thead>${headersHtml}${filterInputsHtml}</thead><tbody>${rows}</tbody></table>`;
  icons();
}

function filterTableBaku(tableId){
  const term = document.getElementById(tableId === 'soPendingTable' ? 'globalSearchSO' : (tableId === 'soOutTable' ? 'globalSearchOut' : 'globalSearchRetur')).value.toLowerCase();
  document.querySelectorAll('#tbl-' + tableId + ' tbody tr.baku-row').forEach(tr=>{
    tr.style.display = tr.textContent.toLowerCase().includes(term) ? '' : 'none';
  });
}

function applyColumnFilter(tableId, colIdx, val){
  const table = document.getElementById('tbl-' + tableId);
  if(!table) return;
  table.querySelectorAll('tbody tr.baku-row').forEach(tr=>{
    const td = tr.querySelectorAll('td')[colIdx];
    if(!td) return;
    const match = val === '' || td.textContent.toLowerCase().includes(val.toLowerCase());
    tr.dataset.filterCol = tr.dataset.filterCol || "{}";
    let filters = JSON.parse(tr.dataset.filterCol);
    filters[colIdx] = match;
    tr.dataset.filterCol = JSON.stringify(filters);
    let show = true; for(let k in filters){ if(!filters[k]) show = false; }
    tr.style.display = show ? '' : 'none';
  });
}

let localChatCache = {};

function refreshChat(){
  if(!currentChatInvoice) return;
  google.script.run.withSuccessHandler(history=>{
    const box = document.getElementById('chatBox');
    if(!history || history.length===0){
      if(localChatCache[currentChatInvoice] && localChatCache[currentChatInvoice].length > 0) { renderChatMessages(localChatCache[currentChatInvoice]); return; }
      box.innerHTML = '<p style="text-align:center;color:var(--text-soft);font-size:13px;margin:30px 0;">Belum ada catatan diskusi.</p>'; return;
    }
    localChatCache[currentChatInvoice] = history;
    renderChatMessages(history);
  }).getFeedbackHistory(currentChatInvoice);
}

function renderChatMessages(history){
  const box = document.getElementById('chatBox');
  box.innerHTML = history.map(h => '<div class="bubble ' + (h.sender.includes(CURRENT_USER.username)?'me':'other') + '"><span class="b-sender">' + escapeHtml(h.sender) + '</span>' + escapeHtml(h.pesan) + '</div>').join('');
  box.scrollTop = box.scrollHeight;
}

function sendFastFeedbackChat(title, messageText){
  if(!currentChatInvoice || !messageText) return;
  const box = document.getElementById('chatBox');
  if(box.querySelector('p')) box.innerHTML = '';
  const senderName = CURRENT_USER.role + ' - ' + CURRENT_USER.username;
  
  if(!localChatCache[currentChatInvoice]) localChatCache[currentChatInvoice] = [];
  localChatCache[currentChatInvoice].push({ sender: senderName, pesan: messageText });
  renderChatMessages(localChatCache[currentChatInvoice]);

  google.script.run.withSuccessHandler(() => {
    google.script.run.withSuccessHandler(history => {
      if(history && history.length > 0) { localChatCache[currentChatInvoice] = history; renderChatMessages(history); }
    }).getFeedbackHistory(currentChatInvoice);
  }).sendFeedbackMessage(currentChatInvoice, senderName, messageText);
}

function exportTableToExcel(tableId, filename = 'Data_Export', maxColLimit = 99) {
  const table = document.getElementById(tableId);
  if (!table) { alert('Data tabel tidak ditemukan untuk diexport!'); return; }
  const clonedTable = table.cloneNode(true);
  const filterRow = clonedTable.querySelector('.filter-row');
  if (filterRow) filterRow.remove();
  clonedTable.querySelectorAll('button, select').forEach(el => el.remove());
  const rows = clonedTable.querySelectorAll('tr');
  rows.forEach(tr => { const cells = tr.querySelectorAll('th, td'); for (let i = maxColLimit; i < cells.length; i++) { cells[i].remove(); } });
  clonedTable.querySelectorAll('td, th').forEach(cell => { let text = cell.textContent.trim(); if (/^\d{10,25}$/.test(text)) { cell.setAttribute('t', 's'); } });
  const wb = XLSX.utils.table_to_book(clonedTable, { sheet: "Sheet1", raw: true });
  XLSX.writeFile(wb, filename + '_' + new Date().toISOString().slice(0,10) + '.xlsx');
}

function handleSingleUpload(targetSheetName, inputEl) {
  if (!inputEl.files || inputEl.files.length === 0) return;
  const file = inputEl.files[0];
  const labelEl = document.getElementById('label_' + targetSheetName);
  const resultEl = document.getElementById('uploadResult');
  labelEl.textContent = file.name;
  resultEl.innerHTML = `Memproses file untuk <b>${targetSheetName}</b>...`;

  const reader = new FileReader();
  reader.onload = e => {
    try {
      const wb = XLSX.read(e.target.result, { type: 'array' });
      const firstSheetName = wb.SheetNames[0];
      const ws = wb.Sheets[firstSheetName];
      const arr = XLSX.utils.sheet_to_json(ws, { header: 1, raw: false, defval: '' });
      if (!arr || arr.length === 0) { resultEl.innerHTML = `<span style="color:var(--danger)">File kosong!</span>`; return; }
      google.script.run.withSuccessHandler(res => {
        if (res.success) { resultEl.innerHTML = `<div class="u-row" style="color:var(--ok); font-weight:600;"><i data-lucide="check-circle" style="width:16px;display:inline;"></i> ${res.message}</div>`; icons(); loadDashboard(); } 
        else { resultEl.innerHTML = `<span style="color:var(--danger)">Gagal: ${res.message}</span>`; }
      }).processArrayUpload(targetSheetName, arr, CURRENT_USER.role);
    } catch (err) { resultEl.innerHTML = `<span style="color:var(--danger)">Error membaca file: ${err.message}</span>`; }
  };
  reader.readAsArrayBuffer(file);
}

// --- FUNGSI SORTING & FILTER GABUNGAN TAB 1 ---
function applyAgingSortAndFilter() {
  let query = document.getElementById('searchAging').value.toLowerCase();
  let sortType = document.getElementById('sortAging').value;
  
  // 1. Filter pencarian
  let filtered = globalMasterSom.filter(d => {
    return d.sku.toLowerCase().includes(query) || 
           d.product.toLowerCase().includes(query) || 
           d.brand.toLowerCase().includes(query);
  });
  
  // 2. Sorting (Pengurutan)
  filtered.sort((a, b) => {
    if (sortType === 'AGING_DESC') return b.agingM - a.agingM; // Terlama ke terbaru
    if (sortType === 'AGING_ASC') return a.agingM - b.agingM;  // Terbaru ke terlama
    if (sortType === 'SKU_AZ') return a.sku.localeCompare(b.sku); // Abjad A-Z
    return 0;
  });
  
  // 3. Tampilkan ulang
  renderSpvAgingTable(filtered);
}

// --- RENDER TAB 2: PENUGASAN ---
function spvPopulateDropdowns(level) {
  let prodSel = document.getElementById('spvFilterProd').value;
  let brandSel = document.getElementById('spvFilterBrand').value;
  
  if (level === 'PROD') {
    let prods = [...new Set(globalMasterSom.map(d => d.product))].sort();
    let opt = '<option value="">-- Semua Product --</option>';
    prods.forEach(p => opt += `<option value="${escapeHtml(p)}">${escapeHtml(p)}</option>`);
    document.getElementById('spvFilterProd').innerHTML = opt;
    document.getElementById('spvFilterBrand').innerHTML = '<option value="">-- Semua Brand --</option>';
    document.getElementById('spvFilterSku').innerHTML = '<option value="">-- Semua SKU --</option>';
  }
  else if (level === 'BRAND') {
    let filtered = prodSel ? globalMasterSom.filter(d => d.product === prodSel) : globalMasterSom;
    let brands = [...new Set(filtered.map(d => d.brand))].sort();
    let opt = '<option value="">-- Semua Brand --</option>';
    brands.forEach(b => opt += `<option value="${escapeHtml(b)}">${escapeHtml(b)}</option>`);
    document.getElementById('spvFilterBrand').innerHTML = opt;
    document.getElementById('spvFilterSku').innerHTML = '<option value="">-- Semua SKU --</option>';
  }
  else if (level === 'SKU') {
    let filtered = globalMasterSom;
    if(prodSel) filtered = filtered.filter(d => d.product === prodSel);
    if(brandSel) filtered = filtered.filter(d => d.brand === brandSel);
    let skus = [...new Set(filtered.map(d => d.sku))].sort();
    let opt = '<option value="">-- Semua SKU --</option>';
    skus.forEach(s => opt += `<option value="${escapeHtml(s)}">${escapeHtml(s)}</option>`);
    document.getElementById('spvFilterSku').innerHTML = opt;
  }
}

function spvHandleFilterChange(levelChanged) {
  if (levelChanged === 'PROD') spvPopulateDropdowns('BRAND');
  if (levelChanged === 'BRAND') spvPopulateDropdowns('SKU');
  
  let p = document.getElementById('spvFilterProd').value;
  let b = document.getElementById('spvFilterBrand').value;
  let s = document.getElementById('spvFilterSku').value;
  
  let filtered = globalMasterSom.filter(d => {
    return (!p || d.product === p) && (!b || d.brand === b) && (!s || d.sku === s);
  });
  spvRenderSomTable(filtered);
}

// Variabel global untuk menyimpan history SOM agar bisa dibuka di Modal Detail & Export
let globalSomHistory = []; 

// --- RENDER TAB 3 (HASIL - GROUPING TANGGAL & PRODUCT) & TAB 4 (SELISIH - EDITABLE) ---
function renderSomSpvHistoryTable(list, containerId) {
  const container = document.getElementById(containerId);
  if(list.length === 0) {
    container.innerHTML = `<div class="empty-state">Belum ada history/data tersimpan.</div>`;
    return;
  }

  // ====================================================
  // TAB HASIL SOM (GROUP BY TGL -> PRODUCT)
  // ====================================================
  if (containerId === 'spvHasilContainer') {
    globalSomHistory = list; 
    let grouped = {};
    
    // Grouping Lapis 1 (Tanggal), Lapis 2 (Product)
    list.forEach(t => {
      let tglOnly = String(t.tglSubmit).split(' ')[0] || 'Tanggal Unknown';
      let prod = t.product || 'LAINNYA';
      
      if(!grouped[tglOnly]) grouped[tglOnly] = {};
      if(!grouped[tglOnly][prod]) grouped[tglOnly][prod] = { totalSku: 0, sys: 0, fisik: 0, items: [] };
      
      grouped[tglOnly][prod].items.push(t);
      grouped[tglOnly][prod].sys += (t.qtySystem || 0);
      grouped[tglOnly][prod].fisik += (t.qtyFisik !== null ? t.qtyFisik : 0);
      grouped[tglOnly][prod].totalSku += 1;
    });

    let finalHtml = '';
    
    // Render HTML per Tanggal
    Object.keys(grouped).forEach(tglStr => {
      finalHtml += `
        <div style="background:#0f172a; color:#fff; padding:10px 16px; font-size:13.5px; font-weight:800; margin-top:16px; border-radius:10px 10px 0 0; display:flex; align-items:center; gap:8px;">
           <i data-lucide="calendar" style="width:16px;height:16px;"></i> Rekap Stock Opname: ${escapeHtml(tglStr)}
        </div>
        <table class="table-baku" style="width:100%; border:1px solid var(--border); border-top:none; margin-bottom:0; border-radius:0 0 10px 10px;">
          <thead>
            <tr>
              <th>KATEGORI PRODUCT</th>
              <th style="text-align:center;">TOTAL (SKU)</th>
              <th style="text-align:center;">TOTAL SISTEM</th>
              <th style="text-align:center;">TOTAL FISIK</th>
              <th style="text-align:center;">TOTAL SELISIH</th>
              <th style="text-align:center;">AKURASI GLOBAL</th>
              <th style="text-align:center;">AKSI</th>
            </tr>
          </thead>
          <tbody>
      `;
      
      Object.keys(grouped[tglStr]).forEach(prod => {
        let g = grouped[tglStr][prod];
        let diff = g.fisik - g.sys;
        let acc = g.sys === 0 ? (g.fisik === 0 ? 100 : 0) : Math.round((g.fisik / g.sys) * 100);
        let accColor = acc === 100 ? 'var(--ok)' : (acc >= 95 ? 'var(--warn)' : 'var(--danger)');
        let accBg = acc === 100 ? '#d1fae5' : (acc >= 95 ? '#fef3c7' : '#fee2e2');

        finalHtml += `
        <tr>
          <td><b>${escapeHtml(prod)}</b></td>
          <td style="text-align:center;">${g.totalSku} SKU</td>
          <td style="text-align:center;">${g.sys}</td>
          <td style="text-align:center; font-weight:800; color:var(--primary);">${g.fisik}</td>
          <td style="text-align:center; font-weight:800; color:${diff === 0 ? 'var(--ok)' : 'var(--danger)'};">${diff}</td>
          <td style="text-align:center;"><span class="pill" style="background:${accBg}; color:${accColor}; font-size:12px;">${acc}%</span></td>
          <td style="text-align:center;">
            <button class="btn btn-outline" style="padding:6px 12px; font-size:11.5px; display:inline-flex;" onclick="openSomDetail('${escapeHtml(prod)}', '${escapeHtml(tglStr)}')"><i data-lucide="eye" style="width:14px;height:14px;"></i> Detail</button>
            <button class="btn btn-outline" style="padding:6px 12px; font-size:11.5px; display:inline-flex; margin-left:4px;" onclick="exportSomProduct('${escapeHtml(prod)}', '${escapeHtml(tglStr)}')"><i data-lucide="file-spreadsheet" style="width:14px;height:14px;"></i> Export</button>
          </td>
        </tr>`;
      });
      finalHtml += `</tbody></table>`;
    });

    container.innerHTML = finalHtml;
    icons();
  } 
  // ====================================================
  // TAB SELISIH SOM (FLAT & FITUR EDIT KOREKSI)
  // ====================================================
  else {
    let rows = list.map(t => {
      let accColor = t.percentage === 100 ? 'var(--ok)' : 'var(--danger)';
      return `
      <tr id="row_som_${t.id}">
        <td>${escapeHtml(t.tglSubmit)}</td>
        <td><b>${escapeHtml(t.picker)}</b><br><span style="font-size:10px; color:var(--text-soft);">${escapeHtml(t.product)}</span></td>
        <td><b>${escapeHtml(t.sku)}</b></td>
        <td style="text-align:center;">${t.qtySystem}</td>
        
        <!-- KOLOM QTY FISIK YANG BISA DI-EDIT -->
        <td style="text-align:center; font-weight:800; color:var(--primary);">
           <span id="txtFisik_${t.id}">${t.qtyFisik}</span>
           <input type="number" id="inpFisik_${t.id}" value="${t.qtyFisik}" style="display:none; width:60px; padding:6px; text-align:center; border:2px solid var(--primary); border-radius:6px; outline:none; font-weight:800;">
        </td>
        
        <td style="text-align:center; font-weight:800; color:${accColor};">${t.diff}</td>
        <td style="text-align:center;"><span class="pill" style="background:#f1f5f9; color:${accColor};">${t.percentage}%</span></td>
        
        <!-- KOLOM AKSI KOREKSI -->
        <td style="text-align:center;">
           <button class="btn btn-outline" id="btnEdit_${t.id}" style="padding:6px 12px; font-size:11px;" onclick="toggleEditSom('${t.id}')"><i data-lucide="edit-3" style="width:14px;height:14px;"></i> Koreksi</button>
           <button class="btn btn-primary" id="btnSave_${t.id}" style="display:none; padding:6px 12px; font-size:11px;" onclick="saveKoreksiSom('${t.id}')"><i data-lucide="save" style="width:14px;height:14px;"></i> Simpan</button>
           <button class="btn btn-outline" id="btnCancel_${t.id}" style="display:none; padding:6px 8px; font-size:11px; margin-left:4px;" onclick="cancelEditSom('${t.id}')"><i data-lucide="x" style="width:14px;height:14px;margin:0;"></i></button>
        </td>
      </tr>
    `}).join('');
    
    container.innerHTML = `
      <table class="table-baku" style="width:100%;">
        <thead>
          <tr><th>WAKTU SUBMIT</th><th>NAMA PICKER</th><th>SKU BARANG</th><th style="text-align:center;">SISTEM</th><th style="text-align:center;">FISIK RAK</th><th style="text-align:center;">SELISIH</th><th style="text-align:center;">AKURASI</th><th style="text-align:center;">AKSI</th></tr>
        </thead>
        <tbody>${rows}</tbody>
      </table>
    `;
    icons();
  }
}

// --- FUNGSI KONTROL EDIT SELISIH ---
function toggleEditSom(id) {
  document.getElementById('txtFisik_' + id).style.display = 'none';
  document.getElementById('btnEdit_' + id).style.display = 'none';
  
  document.getElementById('inpFisik_' + id).style.display = 'inline-block';
  document.getElementById('btnSave_' + id).style.display = 'inline-flex';
  document.getElementById('btnCancel_' + id).style.display = 'inline-flex';
}

function cancelEditSom(id) {
  document.getElementById('txtFisik_' + id).style.display = 'inline-block';
  document.getElementById('btnEdit_' + id).style.display = 'inline-flex';
  
  document.getElementById('inpFisik_' + id).style.display = 'none';
  document.getElementById('btnSave_' + id).style.display = 'none';
  document.getElementById('btnCancel_' + id).style.display = 'none';
}

// Variabel sementara untuk menyimpan ID yang sedang mau dikoreksi
let currentKoreksiId = null;

function saveKoreksiSom(id) {
  let newQty = document.getElementById('inpFisik_' + id).value;
  if(newQty === '') { alert('QTY Fisik tidak boleh kosong!'); return; }
  
  // Simpan ID ke variabel global, lalu panggil pop-up estetiknya
  currentKoreksiId = id;
  document.getElementById('modalConfirmJelembur').classList.add('active');
  icons();
}

// Fungsi yang dieksekusi saat tombol "Ya, Yakin!" di dalam modal ditekan
function executeSaveKoreksi() {
  if (!currentKoreksiId) return;
  
  let id = currentKoreksiId;
  let newQty = document.getElementById('inpFisik_' + id).value;
  
  // Tutup pop-up
  closeModal('modalConfirmJelembur');
  
  // Ubah tombol jadi loading
  let btn = document.getElementById('btnSave_' + id);
  btn.disabled = true; 
  btn.innerHTML = '<i class="spinner"></i>';
  
  // Eksekusi ke backend
  google.script.run.withSuccessHandler(res => {
    if(res.success) {
      document.getElementById('toastMsg').textContent = res.message;
      document.getElementById('toast').classList.add('show');
      setTimeout(()=> document.getElementById('toast').classList.remove('show'), 3000);
      
      currentKoreksiId = null; // Reset
      loadSomSpvData(); // Refresh data tabel
    } else {
      alert(res.message);
      btn.disabled = false; 
      btn.innerHTML = '<i data-lucide="save"></i> Simpan';
      icons();
    }
  }).updateSomFisik(id, newQty);
}

// --- FUNGSI EXPORT KHUSUS PER PRODUCT KE EXCEL ---
function exportSomProduct(product) {
  let items = globalSomHistory.filter(t => t.product === product);
  if (items.length === 0) { alert('Tidak ada data untuk diexport!'); return; }

  // Susun data array untuk library XLSX (Baris pertama adalah Header)
  let exportData = [
    ['WAKTU SUBMIT', 'NAMA PICKER', 'PRODUCT CATEGORY', 'BRAND', 'SKU BARANG', 'QTY SYSTEM', 'QTY FISIK', 'SELISIH QTY', 'AKURASI (%)']
  ];

  // Masukkan baris data
  items.forEach(t => {
    exportData.push([
      t.tglSubmit, t.picker, t.product, t.brand, t.sku, 
      t.qtySystem, t.qtyFisik, t.diff, t.percentage
    ]);
  });

  // Generate File Excel menggunakan library XLSX yang sudah ada
  let ws = XLSX.utils.aoa_to_sheet(exportData);
  let wb = XLSX.utils.book_new();
  
  // Potong nama sheet maksimal 31 karakter (Aturan Excel)
  let safeSheetName = ("SOM_" + product).substring(0, 30);
  XLSX.utils.book_append_sheet(wb, ws, safeSheetName);
  
  // Download File
  let dateStr = new Date().toISOString().slice(0, 10);
  XLSX.writeFile(wb, `Laporan_SOM_${product.replace(/\s+/g, '_')}_${dateStr}.xlsx`);
}

function toggleAllSomCheckbox(chkAll) {
  document.querySelectorAll('.chk-som-item').forEach(chk => chk.checked = chkAll.checked);
}

function assignSomToPicker() {
  const picker = document.getElementById('spvPickerSelect').value;
  if(!picker) { alert('Silakan pilih nama Picker terlebih dahulu!'); return; }
  
  let selectedTasks = [];
  document.querySelectorAll('.chk-som-item:checked').forEach(chk => {
    selectedTasks.push({
      sku: chk.getAttribute('data-sku'),
      product: chk.getAttribute('data-prod'),
      brand: chk.getAttribute('data-brand'),
      qtySystem: chk.getAttribute('data-qty')
    });
  });
  
  if(selectedTasks.length === 0) { alert('Centang minimal 1 barang untuk ditugaskan!'); return; }
  
  const btn = document.getElementById('btnAssignSom');
  btn.disabled = true; btn.innerHTML = '<i class="spinner"></i> Mengirim Tugas...';
  
  google.script.run.withSuccessHandler(res => {
    btn.disabled = false; btn.innerHTML = '<i data-lucide="send"></i> Assign Tugas Terpilih';
    icons();
    if(res.success) {
      document.getElementById('toastMsg').textContent = res.message;
      document.getElementById('toast').classList.add('show');
      setTimeout(()=> document.getElementById('toast').classList.remove('show'), 3000);
      document.querySelectorAll('.chk-som-item').forEach(c => c.checked = false);
      document.getElementById('chkAllSom').checked = false;
    } else alert(res.message);
  }).assignSomTask(picker, selectedTasks);
}

// --- FUNGSI KHUSUS LAYAR PICKER ---
function switchSomPickerTab(tab) {
  ['Pending', 'History', 'Selisih'].forEach(t => {
    document.getElementById('tabBtnSom' + t).classList.remove('active');
    document.getElementById('tabSom' + t).style.display = 'none';
  });
  
  document.getElementById('tabBtnSom' + tab.charAt(0).toUpperCase() + tab.slice(1)).classList.add('active');
  document.getElementById('tabSom' + tab.charAt(0).toUpperCase() + tab.slice(1)).style.display = 'block';
}

function loadSomPickerData() {
  document.getElementById('somPickerPendingContainer').innerHTML = '<div class="empty-state">Menarik data tugas SOM...</div>';
  
  google.script.run.withSuccessHandler(data => {
    renderSomPickerPending(data.pending);
    renderSomPickerTable(data.history, 'somPickerHistoryContainer');
    renderSomPickerTable(data.selisih, 'somPickerSelisihContainer');
    icons();
  }).getSomDataByPicker(CURRENT_USER.username, CURRENT_USER.role);
}

function submitSomTask(taskId) {
  const inputEl = document.getElementById('qtyFisik_' + taskId);
  const btn = document.getElementById('btnSubmit_' + taskId);
  const qty = inputEl.value;
  
  if(qty === '') { alert('Qty Fisik wajib diisi angka (isi 0 jika barang kosong)!'); return; }
  
  btn.disabled = true; 
  btn.innerHTML = '<i class="spinner"></i>';
  
  google.script.run.withSuccessHandler(res => {
    if(res.success) {
      document.getElementById('toastMsg').textContent = 'Hasil hitung fisik berhasil disubmit!';
      document.getElementById('toast').classList.add('show');
      setTimeout(()=> document.getElementById('toast').classList.remove('show'), 3000);
      loadSomPickerData(); // Refresh UI otomatis
    } else {
      alert(res.message);
      btn.disabled = false; btn.innerHTML = 'Submit';
    }
  }).submitSomFisik(taskId, Number(qty));
}
